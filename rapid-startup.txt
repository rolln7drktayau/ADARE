# Rapid startup (Make-based)
# ------------------------------------
# 1) Install/prep environment, then open the menu
make

# 2) If dependencies are already ready, open only the menu
make menu

# 3) Run the default experiment configured in scripts/main.py + config files
make run

# 4) Run ADARE only, without a baseline comparison
make adare
make adare-1000

# 5) Open the evolution dashboard: generation, convergence, metrics, Pareto view
make live

# 6) Run one specific benchmark from the Makefile pattern rule
#    Example targets: Montage_25, CyberShake_30, Epigenomics_24
make run-Montage_25
make run-CyberShake_30
make run-Epigenomics_24

# 7) Optional: clean generated outputs and caches
make clean

# 8) Optional: formatting and linting
make format
make lint

# 9) List available commands
make help

# 10) Test syntax hardening for runnable scripts
make prepare
